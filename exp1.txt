#include <bits/stdc++.h>

using namespace std;

typedef complex<double> cd;

int main() {
    int nb = 3;
    int nl = 3;

    vector<vector<double>> ld = {
        {1, 2, 0.02, 0.06, 0.03},
        {1, 3, 0.08, 0.024, 0.025},
        {2, 3, 0.06, 0.018, 0.02}
    };

    vector<vector<cd>> Y(nb, vector<cd>(nb, cd(0, 0)));

    for (int k = 0; k < nl; k++) {
        int i = ld[k][0] - 1;
        int j = ld[k][1] - 1;

        double r = ld[k][2];
        double x = ld[k][3];
        double b = ld[k][4];

        cd z(r, x);
        cd y = 1.0 / z;
        cd bsh(0, b);

        Y[i][j] = Y[i][j] - y;
        Y[j][i] = Y[i][j];

        Y[i][i] = Y[i][i] + y + bsh;
        Y[j][j] = Y[j][j] + y + bsh;
    }

    cout << "Ybus matrix:\n";
    for (int i = 0; i < nb; i++) {
        for (int j = 0; j < nb; j++) {
            cout << real(Y[i][j]) << " + j" << imag(Y[i][j]) << "\t";
        }
        cout << endl;
    }

    int elim = 2;

    vector<int> keep;
    for (int i = 0; i < nb; i++) {
        if (i != elim) keep.push_back(i);
    }

    vector<vector<cd>> Yaa(2, vector<cd>(2));
    vector<cd> Yab(2), Yba(2);
    cd Ybb = Y[elim][elim];

    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            Yaa[i][j] = Y[keep[i]][keep[j]];
        }
        Yab[i] = Y[keep[i]][elim];
        Yba[i] = Y[elim][keep[i]];
    }

    vector<vector<cd>> Yred(2, vector<cd>(2));

    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            Yred[i][j] = Yaa[i][j] - (Yab[i] * Yba[j]) / Ybb;
        }
    }

    cout << "\nReduced Ybus matrix:\n";
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            cout << real(Yred[i][j]) << " + j" << imag(Yred[i][j]) << "\t";
        }
        cout << endl;
    }

    return 0;
}
